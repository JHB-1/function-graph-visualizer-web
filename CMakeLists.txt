cmake_minimum_required(VERSION 3.24)
project(FunctionGraphVisualizerWeb VERSION 0.1.0 LANGUAGES CXX)
set(FGV_CORE_SOURCE "" CACHE PATH "Pinned private core checkout")
set(FGV_JSON_INCLUDE "" CACHE PATH "nlohmann/json.hpp parent directory")
if(NOT TARGET fgv::core)
	if(NOT EXISTS "${FGV_CORE_SOURCE}/CMakeLists.txt")
		message(FATAL_ERROR "Set FGV_CORE_SOURCE to the private core checkout; see core.lock.json")
	endif()
	set(FGV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
	set(FGV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
	set(FGV_BUILD_PLUGIN_HOST OFF CACHE BOOL "" FORCE)
	add_subdirectory("${FGV_CORE_SOURCE}" core)
endif()
if(NOT FGV_JSON_INCLUDE)
	find_package(Python3 REQUIRED COMPONENTS Interpreter)
	execute_process(COMMAND "${Python3_EXECUTABLE}" -c
		"import nlohmann_json; print(nlohmann_json.get_include())"
		OUTPUT_VARIABLE FGV_JSON_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endif()
add_library(fgv_web_bridge STATIC native/bridge.cpp native/operations.cpp)
target_include_directories(fgv_web_bridge PUBLIC native "${FGV_JSON_INCLUDE}")
target_link_libraries(fgv_web_bridge PUBLIC fgv::core)
target_compile_features(fgv_web_bridge PUBLIC cxx_std_23)
target_compile_options(fgv_web_bridge PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
if(EMSCRIPTEN)
	target_compile_options(function_graph_core PRIVATE -fexceptions)
	target_compile_options(fgv_web_bridge PRIVATE -fexceptions)
	add_executable(fgv_runtime native/wasm.cpp)
	target_link_libraries(fgv_runtime PRIVATE fgv_web_bridge)
	target_compile_options(fgv_runtime PRIVATE -fexceptions)
	target_link_options(fgv_runtime PRIVATE -fexceptions
		"-sEXPORTED_FUNCTIONS=['_fgv_request','_malloc','_free']"
		"-sEXPORTED_RUNTIME_METHODS=['ccall']" -sMODULARIZE=1 -sEXPORT_ES6=1
		-sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=1073741824 -sSTACK_SIZE=1048576
		-sENVIRONMENT=web,worker,node -sFILESYSTEM=0)
	set_target_properties(fgv_runtime PROPERTIES OUTPUT_NAME fgv-runtime SUFFIX .mjs)
else()
	add_executable(fgv_plugin_host native/main.cpp)
	target_link_libraries(fgv_plugin_host PRIVATE fgv_web_bridge)
	install(TARGETS fgv_plugin_host RUNTIME DESTINATION bin)
endif()
